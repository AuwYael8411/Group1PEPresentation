<script>

const canvas = document.getElementById("maze");
const ctx = canvas.getContext("2d");

const message = document.getElementById("message");
const answer = document.getElementById("answer");
const continueBtn = document.getElementById("continueBtn");


// ===============================
// MAZE SETTINGS
// ===============================

const cellSize = 50;

const maze = [

    "#################",
    "#S              #",
    "# ############# #",
    "# #             #",
    "# # #############",
    "# #             #",
    "# ############# #",
    "#               #",
    "############### #",
    "#               #",
    "# ###############",
    "#               #",
    "############### #",
    "#               #",
    "# ############# #",
    "#             G #",
    "#################"

];

const rows = maze.length;
const cols = maze[0].length;

canvas.width = cols * cellSize;
canvas.height = rows * cellSize;


// ===============================
// PLAYER
// ===============================

let player = {
    x: 1.5 * cellSize,
    y: 1.5 * cellSize,
    radius: 8
};

let gameFinished = false;


// ===============================
// DRAW MAZE
// ===============================

function drawMaze() {

    ctx.clearRect(
        0,
        0,
        canvas.width,
        canvas.height
    );


    // Draw walls

    for (let row = 0; row < rows; row++) {

        for (let col = 0; col < cols; col++) {

            if (maze[row][col] === "#") {

                ctx.fillStyle = "#000";

                ctx.fillRect(
                    col * cellSize,
                    row * cellSize,
                    cellSize,
                    cellSize
                );

            }

        }

    }


    // Draw START

    ctx.fillStyle = "#ff3d3d";

    ctx.fillRect(
        cellSize,
        cellSize,
        cellSize,
        cellSize
    );


    ctx.fillStyle = "white";

    ctx.font = "20px Pixelify Sans";

    ctx.fillText(
        "START",
        cellSize + 5,
        cellSize + 30
    );


    // Draw GOAL

    for (let row = 0; row < rows; row++) {

        for (let col = 0; col < cols; col++) {

            if (maze[row][col] === "G") {

                ctx.fillStyle = "#ff3d3d";

                ctx.fillRect(
                    col * cellSize,
                    row * cellSize,
                    cellSize,
                    cellSize
                );

                ctx.fillStyle = "white";

                ctx.font = "22px Pixelify Sans";

                ctx.fillText(
                    "GO!",
                    col * cellSize + 7,
                    row * cellSize + 32
                );

            }

        }

    }


    // Draw player

    ctx.beginPath();

    ctx.arc(
        player.x,
        player.y,
        player.radius,
        0,
        Math.PI * 2
    );

    ctx.fillStyle = "#fff";

    ctx.fill();

    ctx.strokeStyle = "#000";

    ctx.lineWidth = 3;

    ctx.stroke();

}


// ===============================
// CHECK WALL COLLISION
// ===============================

function touchingWall(x, y) {

    const left =
        Math.floor((x - player.radius) / cellSize);

    const right =
        Math.floor((x + player.radius) / cellSize);

    const top =
        Math.floor((y - player.radius) / cellSize);

    const bottom =
        Math.floor((y + player.radius) / cellSize);


    for (let row = top; row <= bottom; row++) {

        for (let col = left; col <= right; col++) {

            if (
                row < 0 ||
                row >= rows ||
                col < 0 ||
                col >= cols
            ) {

                return true;

            }


            if (maze[row][col] === "#") {

                return true;

            }

        }

    }

    return false;

}


// ===============================
// CHECK GOAL
// ===============================

function reachedGoal(x, y) {

    const col = Math.floor(x / cellSize);
    const row = Math.floor(y / cellSize);

    return maze[row][col] === "G";

}


// ===============================
// RESET PLAYER
// ===============================

function resetPlayer() {

    player.x = 1.5 * cellSize;
    player.y = 1.5 * cellSize;

    message.textContent = "OUCH! DON'T TOUCH THE WALLS.";

    setTimeout(() => {

        if (!gameFinished) {
            message.textContent = "TRY AGAIN :)";
        }

    }, 1000);

    drawMaze();

}


// ===============================
// MOUSE MOVEMENT
// ===============================

canvas.addEventListener("mousemove", function(event) {

    if (gameFinished) {
        return;
    }


    const rect = canvas.getBoundingClientRect();


    const mouseX =
        event.clientX - rect.left;

    const mouseY =
        event.clientY - rect.top;


    // Check collision before moving

    if (touchingWall(mouseX, mouseY)) {

        resetPlayer();

        return;

    }


    player.x = mouseX;
    player.y = mouseY;


    // Check goal

    if (reachedGoal(player.x, player.y)) {

        finishGame();

        return;

    }


    drawMaze();

});


// ===============================
// FINISH GAME
// ===============================

function finishGame() {

    gameFinished = true;

    message.textContent =
        "YOU MADE IT!";

    answer.style.display = "block";

    continueBtn.style.display =
        "inline-block";

    drawMaze();

}


// ===============================
// START GAME
// ===============================

drawMaze();

</script>
